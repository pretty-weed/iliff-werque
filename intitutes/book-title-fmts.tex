\newcommand{\marginheading}[1]{%
	\marginnote{\normalfont\itshape #1}[-4.75ex]%
}

\titleformat{\subsection}[runin]
{\normalfont\bfseries} % Format for the inline number (e.g., Bold)
{\thesubsection}       % Position the number as the inline run-in anchor
{0.5em}                % Space between the number and the start of paragraph text
{\marginheading}

% 3. Adjust the vertical spacing around your new subsection entries
\titlespacing*{\subsection}
{0pt}                         % Indent before the number
{3.25ex plus 1ex minus .2ex}  % Space before the paragraph starts
{0.5em}

% 3. Adjust the vertical spacing around your new subsection entries
\titlespacing*{\paragraph}
